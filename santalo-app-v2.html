function aggiornaPulizie() { var preview = document.getElementById('puliziePreview'); if (!preview) return; var target = new Date(); target.setHours(0,0,0,0); if (pulDay === 'domani') target.setDate(target.getDate() + 1); var targetStr = target.toISOString().slice(0,10); var targetFmt = target.toLocaleDateString('it-IT', {weekday:'long', day:'numeric', month:'long'}); var checkouts = prenotazioni.filter(function(p) { return p.partenza === targetStr; }); if (!checkouts.length) { preview.textContent = 'Nessun check-out ' + (pulDay==='oggi'?'oggi':'domani') + ' - nessuna camera da pulire.'; return; } var righe = checkouts.map(function(p) { var cam = (p.camera||'').replace('Camera ',''); var ospite = (p.nome||'').split(' ')[0]; var nuovoArrivo = prenotazioni.find(function(p2) { return p2.arrivo === targetStr && p2.camera === p.camera && p2.id !== p.id; }); var riga = String.fromCharCode(8226) + ' Camera ' + cam + ' - check-out ' + ospite; if (nuovoArrivo) { riga += ' Attenzione: nuovo arrivo stesso giorno: ' + (nuovoArrivo.nome||'').split(' ')[0]; } return riga; }); var quante = checkouts.length === 1 ? '1 camera da rifare' : checkouts.length + ' camere da rifare'; var quando = pulDay === 'oggi' ? 'oggi' : 'domani'; var msg = 'Ciao Marianna! '; msg += 'Per ' + quando + ' (' + targetFmt + ') ho ' + quante + ': '; msg += righe.join(' '); msg += ' Grazie!'; preview.textContent = msg; }
